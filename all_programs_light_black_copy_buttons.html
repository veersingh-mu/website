Question 1 — Convert decimal/hexadecimal to binary (Turbo C)

#include <stdio.h>
#include <conio.h>

void main()
{
    unsigned long num;
    int base, i = 0, j;
    char bin[33];

    clrscr();
    printf("Enter base (10=decimal, 16=hex): ");
    scanf("%d", &base);

    if (base == 10)
    {
        printf("Enter decimal number: ");
        scanf("%lu", &num);
    }
    else if (base == 16)
    {
        printf("Enter hexadecimal number (without 0x): ");
        scanf("%lx", &num);
    }
    else
    {
        printf("Invalid base!");
        getch();
        return;
    }

    if (num == 0)
        bin[i++] = '0';

    while (num > 0 && i < 32)
    {
        bin[i++] = (num % 2) + '0';
        num = num / 2;
    }

    printf("\\nBinary number: ");
    for (j = i - 1; j >= 0; j--)
        printf("%c", bin[j]);

    getch();
}

Question 2 — 2's complement of an 8-bit number (Turbo C)

#include <stdio.h>
#include <conio.h>

void main()
{
    unsigned int num, result;
    int i;

    clrscr();
    printf("Enter an 8-bit number (0-255): ");
    scanf("%u", &num);

    if (num > 255)
    {
        printf("Invalid input!");
        getch();
        return;
    }

    result = ((~num) & 0xFF) + 1;
    result = result & 0xFF;

    printf("\\nOriginal binary: ");
    for (i = 7; i >= 0; i--)
        printf("%d", (num >> i) & 1);

    printf("\\n2's complement: ");
    for (i = 7; i >= 0; i--)
        printf("%d", (result >> i) & 1);

    printf("\\nUnsigned result: %u", result);
    getch();
}

Experiment 8 — Bubble sort 8-bit numbers in ascending order

org 100h

    jmp START

ARR db 5, 2, 9, 1, 7, 6, 3, 8
N equ 8

START:
    mov cx, N-1

OUTER_LOOP:
    push cx
    lea si, ARR
    mov cx, N-1

INNER_LOOP:
    mov al, [si]
    mov ah, [si+1]
    cmp al, ah
    jbe NO_SWAP
    mov [si], ah
    mov [si+1], al

NO_SWAP:
    inc si
    loop INNER_LOOP

    pop cx
    loop OUTER_LOOP

    lea si, ARR
    mov cx, N

PRINT_LOOP:
    mov dl, [si]
    add dl, '0'
    mov ah, 02h
    int 21h

    mov dl, ' '
    mov ah, 02h
    int 21h

    inc si
    loop PRINT_LOOP

    mov ah, 4Ch
    int 21h

ret

VS Code / Online Compiler — Question 1: Decimal or hexadecimal to binary (Standard C)

#include <stdio.h>
#include <stdlib.h>
#include <errno.h>

int main(void)
{
    unsigned long number;
    char input[100];
    char *end;
    int base, i = 0, j;
    char binary[sizeof(unsigned long) * 8 + 1];

    printf("Choose input type (10 = decimal, 16 = hexadecimal): ");
    if (scanf("%d", &base) != 1 || (base != 10 && base != 16)) {
        printf("Invalid base. Choose 10 or 16.\\n");
        return 1;
    }

    printf("Enter the number: ");
    if (scanf("%99s", input) != 1) {
        printf("Invalid input.\\n");
        return 1;
    }

    errno = 0;
    number = strtoul(input, &end, base);
    if (errno != 0 || end == input || *end != '\\0') {
        printf("Invalid number.\\n");
        return 1;
    }

    if (number == 0) {
        binary[i++] = '0';
    } else {
        while (number > 0) {
            binary[i++] = (char)('0' + (number % 2));
            number /= 2;
        }
    }
    binary[i] = '\\0';

    printf("Binary number: ");
    for (j = i - 1; j >= 0; j--) {
        putchar(binary[j]);
    }
    putchar('\\n');

    return 0;
}

VS Code / Online Compiler — Question 2: 8-bit two's complement (Standard C)

#include <stdio.h>

int main(void)
{
    unsigned int number, result;
    int i;

    printf("Enter an 8-bit number (0-255): ");
    if (scanf("%u", &number) != 1 || number > 255) {
        printf("Invalid input. Enter a number from 0 to 255.\\n");
        return 1;
    }

    result = ((~number) & 0xFFu) + 1u;
    result &= 0xFFu;

    printf("Original binary: ");
    for (i = 7; i >= 0; i--) {
        printf("%u", (number >> i) & 1u);
    }

    printf("\\n2's complement: ");
    for (i = 7; i >= 0; i--) {
        printf("%u", (result >> i) & 1u);
    }

    printf("\\nUnsigned 8-bit result: %u\\n", result);
    return 0;
}
