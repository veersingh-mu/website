Experiment 01

Insertion Sort and Comparative Analysis

Insertion Sort builds a sorted part of the array one item at a time by shifting larger values and inserting the next item in its correct position. Best case is O(n); average and worst cases are O(n^2). It is stable and uses O(1) extra space.

Start
Declare an array a[50] and variables n, i, j, key.
Read the number of elements n.
Read n elements into the array.
Set j = 1 and repeat while j < n:
Store a[j] in key.
Set i = j - 1.
Compare the key with the previous elements.
While i >= 0 and a[i] > key:
Shift a[i] one position to the right.
Decrease i by 1.
Insert key at position i + 1.
Increment j.
Display the sorted array.
Stop.
#include <stdio.h>
int main() {
int a[50],n,i,j,key;
printf("Enter the number elements: ");
scanf("%d",&n);
printf("Enter the elements: \n");
for(i=0;i<n;i++) {
scanf("%d",&a[i]);
}
for(j=1;j<n;j++) {
key=a[j];
i=j-1;{
while(i>=0&& a[i]>key)
{
a[i+1]=a[i];
i=i-1;
}
a[i+1]=key;
}}
printf("Sorted Array: \n");
for(i=0;i<n;i++)
{
printf("%d\n",a[i]);
}
return 0;
}

Insertion Sort takes O(n) time in the best case and O(n^2) in average and worst cases.

Experiment 02

Selection Sort and Comparative Analysis

Selection Sort repeatedly finds the smallest value in the unsorted part and swaps it into place. It performs O(n^2) comparisons in all cases, uses O(1) extra space, and is usually not stable.

Start
Declare variables i, j, n, min, temp and an array a[10].
Read the number of elements n.
Read n elements into the array.
Set i = 0 and repeat until i = n - 2:
Assume the current position i contains the minimum element: min = i.
Set j = i + 1.
Compare a[j] with a[min] for all remaining elements:
If a[j] < a[min], set min = j.
If min != i, swap a[i] and a[min].
Repeat the above process for all positions in the array.
Display the sorted array.
Stop.
#include <stdio.h>
int main()
{
int i,j,n,min,a[10],temp;
printf("Enter the number of elements: ");
scanf("%d",&n);
printf("Enter the array: \n");
for(i=0;i<n;i++){
scanf("%d",&a[i]);
}
for(i=0;i<=n-2;i++)
{
min=i;
for (j=i+1;j<=n-1;j++)
{
if (a[j]<a[min])
min=j;
}
if(min!=i)
{
temp=a[i];
a[i]=a[min];
a[min]=temp;
}}
printf("Sorted Array: \n");
for(i=0;i<n;i++)
{
printf("%d\n",a[i]);
}
return 0;
}

Selection Sort takes O(n^2) time in the best, average, and worst cases.

Experiment 03

Binary Search Using Divide and Conquer

Binary Search works on a sorted array by checking the middle element and discarding half of the remaining range. Its best case is O(1), and its average and worst cases are O(log n).

Start
Read the number of elements n.
Read the n elements into array a[].
Read the element key to be searched.
Set:
lb = 0 (lower bound)
ub = n - 1 (upper bound)
Call recBinarySearch(a, ub, lb, key).
In the recursive function:
If lb == ub:
If a[lb] == key, return lb.
Otherwise, return -1.
Calculate mid = (ub + lb) / 2.
If a[mid] == key, return mid.
If key < a[mid], recursively search the left half:
recBinarySearch(a, mid - 1, lb, key)
Otherwise, recursively search the right half:
recBinarySearch(a, ub, mid + 1, key)
If the returned index is not -1, display "Element found" and its index.
Otherwise, display "Element not found".
Stop
#include <stdio.h>

int recBinarySearch(int a[], int ub, int lb, int key) {
    if (lb > ub) {
        return -1;
    }
    int mid = (ub + lb) / 2;
    if (a[mid] == key) {
        return mid;
    }
    if (key < a[mid]) {
        return recBinarySearch(a, mid - 1, lb, key);
    }
    return recBinarySearch(a, ub, mid + 1, key);
}

int main() {
    int a[100], n, key;
    printf("Enter the number of elements: ");
    scanf("%d", &n);
    if (n <= 0 || n > 100) {
        printf("Invalid number of elements.\n");
        return 0;
    }
    printf("Enter the sorted elements: ");
    for (int i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }
    printf("Enter the element to be searched: ");
    scanf("%d", &key);
    int index = recBinarySearch(a, n - 1, 0, key);
    if (index != -1) {
        printf("Element found at index %d\n", index);
    } else {
        printf("Element not found\n");
    }
    return 0;
}

Binary Search takes O(log n) time on a sorted array, with O(1) best-case time.

Experiment 04

Minimum and Maximum Using Divide and Conquer

Divide and Conquer finds the minimum and maximum in each half, then combines the results. It takes O(n) time and O(log n) recursion space for balanced divisions.

Function MinMax(a, low, high)
If low == high, return a[low] as both minimum and maximum.
If high == low + 1:
Compare a[low] and a[high].
Return the smaller value as minimum and the larger value as maximum.
Calculate mid = (low + high) / 2.
Set (min1, max1) = MinMax(a, low, mid).
Set (min2, max2) = MinMax(a, mid + 1, high).
Set minimum = smaller of min1 and min2.
Set maximum = larger of max1 and max2.
Return (minimum, maximum).
Main Algorithm
Start.
Read n and the array elements.
Call MinMax(a, 0, n - 1).
Display the minimum and maximum values.
Stop.
#include <stdio.h>

struct MinMax {
    int minimum;
    int maximum;
    int comparisons;
};

struct MinMax findMinMax(int a[], int low, int high) {
    struct MinMax result;
    if (low == high) {
        result.minimum = a[low];
        result.maximum = a[low];
        result.comparisons = 0;
        return result;
    }
    if (high == low + 1) {
        result.comparisons = 1;
        if (a[low] < a[high]) {
            result.minimum = a[low];
            result.maximum = a[high];
        } else {
            result.minimum = a[high];
            result.maximum = a[low];
        }
        return result;
    }
    int mid = low + (high - low) / 2;
    struct MinMax left = findMinMax(a, low, mid);
    struct MinMax right = findMinMax(a, mid + 1, high);
    result.comparisons = left.comparisons + right.comparisons;
    if (left.minimum < right.minimum) result.minimum = left.minimum;
    else result.minimum = right.minimum;
    result.comparisons++;
    if (left.maximum > right.maximum) result.maximum = left.maximum;
    else result.maximum = right.maximum;
    result.comparisons++;
    return result;
}

int main() {
    int a[100], n;
    printf("Enter the number of elements: ");
    scanf("%d", &n);
    if (n <= 0 || n > 100) {
        printf("Enter a value of n from 1 to 100.\n");
        return 0;
    }
    printf("Enter the elements: ");
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    struct MinMax answer = findMinMax(a, 0, n - 1);
    printf("Minimum element: %d\n", answer.minimum);
    printf("Maximum element: %d\n", answer.maximum);
    printf("Number of comparisons: %d\n", answer.comparisons);
    return 0;
}

The Divide and Conquer method finds both values in O(n) time.

Experiment 05

Naive String Matching Algorithm

Naive String Matching checks the pattern at every possible position in the text. Its worst-case time is O(nm), where n is the text length and m is the pattern length; extra space is O(1), excluding output.

Start.
Read the text T and the pattern P.
Let n be the text length and m be the pattern length.
For every shift s from 0 through n - m:
Compare the pattern with the text starting at T[s].
If any character differs, stop that comparison and try the next shift.
If all m characters match, display the shift s.
Continue until all valid shifts have been checked.
Stop.
#include <stdio.h>
#include <string.h>
void naiveStringMatch(char* T, char* P) {
int n = strlen(T);
int m = strlen(P);
for (int s = 0; s <= n - m; s++) {
int j;
for (j = 0; j < m; j++) {
if (T[s + j] != P[j]) {
break;
}
}
if (j == m) {
printf("Pattern occurs at shift %d\n", s);
}
}
}
int main() {
char T[] = "AABAACAADAABAABA";
char P[] = "AABA";
naiveStringMatch(T, P);
return 0;
}

Naive String Matching has O(nm) worst-case time.

Experiment 06

Sum of Subsets Using Backtracking

Backtracking tries including or excluding each item and stops exploring branches that cannot reach the target. Its worst-case time is O(2^n) and recursion space is O(n); pruning assumes non-negative values.

SumOfSubsets(i, currentSum)
If currentSum == target:
Print the selected subset.
Return.
If i == n OR currentSum > target:
Return.
Include A[i].
Mark A[i] as selected.
SumOfSubsets(i + 1, currentSum + A[i])
Exclude A[i].
Unmark A[i].
SumOfSubsets(i + 1, currentSum)
#include <stdio.h>
void SumOfSubsets(int A[], int selected[], int n, int target, int i, int currentSum) {
if (currentSum == target) {
for (int j = 0; j < i; j++) {
if (selected[j]) {
printf("%d ", A[j]);
}
}
printf("\n");
return;
}
if (i == n || currentSum > target) {
return;
}
selected[i] = 1;
SumOfSubsets(A, selected, n, target, i + 1, currentSum + A[i]);
selected[i] = 0;
SumOfSubsets(A, selected, n, target, i + 1, currentSum);
}
int main() {
int A[] = {10, 7, 5, 18, 12, 20, 15};
int target = 35;
int n = sizeof(A) / sizeof(A[0]);
int selected[100] = {0};
SumOfSubsets(A, selected, n, target, 0, 0);
return 0;
}

Backtracking can find all target-sum subsets; pruning avoids branches that cannot reach the target.

Experiment 07

Fractional Knapsack Using Greedy Method

Fractional Knapsack sorts items by value-to-weight ratio and takes the highest-ratio items first, allowing fractions. This greedy method is optimal for fractional items and takes O(n log n) time because of sorting.

Function GreedyKnapsack(weights, values, capacity):
n = length(weights)
for i = 0 to n-1:
ratio[i] = values[i] / weights[i]
sort items by ratio in descending order
total_value = 0
for i = 0 to n-1:
if capacity == 0:
break
if item weight <= capacity:
take full item
total_value += values[i]
capacity -= weights[i]
else:
take fraction of item
total_value += (values[i] * (capacity / weights[i]))
capacity = 0
return total_value
#include <stdio.h>
#include <stdlib.h>
struct Item {
double weight;
double value;
};
int compare(const void *a, const void *b) {
struct Item *item1 = (struct Item *)a;
struct Item *item2 = (struct Item *)b;
double ratio1 = item1->value / item1->weight;
double ratio2 = item2->value / item2->weight;
if (ratio1 < ratio2) return 1;
else if (ratio1 > ratio2) return -1;
else return 0;
}
double GreedyKnapsack(struct Item items[], int n, double capacity) {
qsort(items, n, sizeof(struct Item), compare);
double total_value = 0.0;
for (int i = 0; i < n; i++) {
if (capacity == 0) {
break;
}
if (items[i].weight <= capacity) {
total_value += items[i].value;
capacity -= items[i].weight;
} else {
total_value += (items[i].value * (capacity / items[i].weight));
capacity = 0;
}
}
return total_value;
}
int main() {
struct Item items[] = {
{10, 60},
{20, 100},
{30, 120}
};
int n = sizeof(items) / sizeof(items[0]);
double capacity = 50.0;
double max_value = GreedyKnapsack(items, n, capacity);
printf("Maximum value we can obtain = %.2f\n", max_value);
return 0;
}

The greedy method is optimal for Fractional Knapsack and takes O(n log n) time.
